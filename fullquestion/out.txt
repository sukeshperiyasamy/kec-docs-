Algorithm:
Start 
Initialize time = 1500 seconds 
Display time in minutes:seconds 
Start timer using setInterval() 
Decrease time every second 
Add Pause and Reset functions 
Stop when time reaches zero 
Stop 
Program:
<!DOCTYPE html>
<html>
<body>

<h1 id="time">25:00</h1>

<button onclick="startTimer()">Start</button>
<button onclick="pauseTimer()">Pause</button>
<button onclick="resetTimer()">Reset</button>

<script>
let time=1500;
let interval;

function updateDisplay(){
let m=Math.floor(time/60);
let s=time%60;
document.getElementById("time").innerText=
m+":"+(s<10?"0":"")+s;
}

function startTimer(){
if(!interval){
interval=setInterval(()=>{
if(time>0){
time--;
updateDisplay();
}
},1000);
}
}

function pauseTimer(){
clearInterval(interval);
interval=null;
}

function resetTimer(){
time=1500;
updateDisplay();
pauseTimer();
}

updateDisplay();
</script>

</body>
</html>
Expected Output:
Timer counts down from 25:00 with Start, Pause, and Reset functionality.
